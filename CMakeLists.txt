cmake_minimum_required(VERSION 3.20)
project(meteora_desktop VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network DBus)
find_package(X11 REQUIRED COMPONENTS Xi)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
qt_standard_project_setup()
add_library(meteora-core STATIC src/core/preferences.cpp src/core/appcatalog.cpp src/core/windowbridge.cpp)
target_include_directories(meteora-core PUBLIC src)
target_compile_definitions(meteora-core PRIVATE METEORA_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_link_libraries(meteora-core PUBLIC Qt6::Widgets PRIVATE PkgConfig::GIO X11::X11 X11::Xi)
target_compile_options(meteora-core PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(meteora-shell src/shell/main.cpp src/shell/desktop.cpp src/shell/notifications.cpp)
target_link_libraries(meteora-shell PRIVATE meteora-core Qt6::Network Qt6::DBus)
target_compile_options(meteora-shell PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(meteora-settings src/settings.cpp)
target_link_libraries(meteora-settings PRIVATE meteora-core)
target_compile_options(meteora-settings PRIVATE -Wall -Wextra -Wpedantic)
option(METEORA_BUILD_LEGACY_PREVIEW "Build the older visual-only Qt prototype" OFF)
if(METEORA_BUILD_LEGACY_PREVIEW)
    qt_add_executable(meteora-preview src/main.cpp)
    target_link_libraries(meteora-preview PRIVATE Qt6::Widgets)
endif()
install(TARGETS meteora-shell meteora-settings RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/meteora-settings.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/meteora-wallpaper.png assets/meteora.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/meteora)
install(DIRECTORY plasma/icons/Meteora DESTINATION ${CMAKE_INSTALL_DATADIR}/icons)
install(DIRECTORY session/themes/Meteora DESTINATION ${CMAKE_INSTALL_DATADIR}/themes)
install(FILES session/openbox.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/meteora)
install(PROGRAMS session/meteora-session DESTINATION ${CMAKE_INSTALL_BINDIR})
# Install the login entry only for an explicit system installation.
if(NOT CMAKE_INSTALL_PREFIX MATCHES "^/home/|^/root/")
    install(FILES session/meteora.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/xsessions)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(render-demo tests/render_demo.cpp src/shell/desktop.cpp src/shell/notifications.cpp)
    target_link_libraries(render-demo PRIVATE meteora-core Qt6::DBus)
    qt_add_executable(test-core tests/test_core.cpp)
    target_link_libraries(test-core PRIVATE meteora-core Qt6::Test)
    add_test(NAME core COMMAND test-core)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
