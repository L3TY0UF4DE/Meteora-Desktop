cmake_minimum_required(VERSION 3.20)
project(meteora_desktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Gui)
qt_standard_project_setup()
qt_add_executable(meteora-preview src/main.cpp)
target_link_libraries(meteora-preview PRIVATE Qt6::Widgets Qt6::Gui)
qt_add_executable(meteora-settings src/settings.cpp)
target_link_libraries(meteora-settings PRIVATE Qt6::Widgets Qt6::Gui)
install(TARGETS meteora-preview meteora-settings RUNTIME DESTINATION bin)
install(FILES packaging/meteora-settings.desktop DESTINATION share/applications)
install(FILES assets/meteora-wallpaper.png DESTINATION share/meteora)
